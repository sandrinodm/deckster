```ts title="webhooks.ts" {2-3|5-8} lineNumbers /verify/ /drop.delivered/
app.post('/hooks', async (req, res) => {
  const signature = req.header('wren-signature');
  if (!wren.webhooks.verify(req.body, signature)) return res.sendStatus(401);

  const event = JSON.parse(req.body);
  if (event.type === 'drop.delivered') {
    await orders.markDelivered(event.drop.reference);
  }
  res.sendStatus(200);
});
```
