```ts title="checkout.ts" {4-7|4-10}
import { Wrenhaul } from '@wrenhaul/sdk';

const wren = new Wrenhaul({ apiKey: process.env.WREN_KEY });
const drop = await wren.drops.create({
  pickup: 'hub:old-town',
  dropoff: '14 Weir Street',
});

drop.status;   // 'batching'
drop.doorEta;  // '09:24', inside the same-hour window
```
